```go
package scenario

import (
	"bytes"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"sort"
	"strings"
	"unicode"
)

func Normalize(id string) string {
	var b strings.Builder
	for _, r := range strings.ToLower(id) {
		if unicode.IsLetter(r) || unicode.IsDigit(r) {
			b.WriteRune(r)
		}
	}
	return b.String()
}

func CanonicalJSON(s []Scenario) ([]byte, error) {
	cp := make([]Scenario, len(s))
	copy(cp, s)
	sort.SliceStable(cp, func(i, j int) bool { return cp[i].ID < cp[j].ID })
	for i := range cp {
		cp[i] = withNonNilSlices(cp[i])
	}
	var buf bytes.Buffer
	enc := json.NewEncoder(&buf)
	enc.SetEscapeHTML(false)
	enc.SetIndent("", "  ")
	if err := enc.Encode(cp); err != nil {
		return nil, err
	}
	return buf.Bytes(), nil
}

func Fingerprint(s []Scenario) (string, error) {
	b, err := CanonicalJSON(s)
	if err != nil {
		return "", err
	}
	sum := sha256.Sum256(b)
	return hex.EncodeToString(sum[:]), nil
}

// withNonNilSlices makes nil and empty slices encode identically ("[]").
func withNonNilSlices(s Scenario) Scenario {
	if s.Path == nil {
		s.Path = []string{}
	}
	if s.Guards == nil {
		s.Guards = []Guard{}
	}
	s.Preconditions = nonNilOutcome(s.Preconditions)
	if s.Stimuli == nil {
		s.Stimuli = []Stimulus{}
	}
	for i := range s.Stimuli {
		s.Stimuli[i].Outcome = nonNilOutcome(s.Stimuli[i].Outcome)
	}
	return s
}

func nonNilOutcome(o Outcome) Outcome {
	if o.ExpectedOutputs == nil {
		o.ExpectedOutputs = []Output{}
	}
	if o.ExpectedStates == nil {
		o.ExpectedStates = []State{}
	}
	if o.Hints == nil {
		o.Hints = []string{}
	}
	return o
}
```

```go
package scenario

type class int

const (
	classInternal class = iota
	classInput
	classOutput
	classState
	classHint
	classStructural
)

// classify applies spec §2.2 in priority order.
func classify(n Node, in Input, view *View) class {
	switch n.Kind {
	case "acceptEvent", "timeEvent":
		return classInput
	case "sendSignal":
		return classOutput
	case "objectNode":
		return classState
	case "note":
		if n.Anchor != "" {
			return classHint
		}
		return classStructural
	case "start", "end", "merge", "fork", "join", "swimLane", "goto", "loop", "interruptEdge":
		return classStructural
	case "decision", "switch":
		if n.DecidedBy != "" && in.Actors[n.DecidedBy] {
			return classInput
		}
		return classInternal
	case "action":
		if n.LinkedActorID != "" && in.Actors[n.LinkedActorID] {
			return classInput
		}
		if view != nil {
			for _, st := range view.Steps {
				if st.NodeID != n.ID {
					continue
				}
				for _, c := range st.Calls {
					if in.Actors[c.To] || (!in.Actors[c.From] && !isComponent(in, c.To)) {
						return classOutput
					}
				}
			}
		}
		return classInternal
	}
	return classInternal
}

func isComponent(in Input, id string) bool {
	_, ok := in.Contracts[Normalize(id)]
	if ok {
		return true
	}
	for k := range in.Contracts {
		if Normalize(k) == Normalize(id) {
			return true
		}
	}
	return false
}

// firstCallInto returns the first call of view step nodeID whose To is a component,
// with its index, for provenance. ok=false when none.
func firstCallInto(in Input, view *View, nodeID string) (c Call, idx int, ok bool) {
	if view == nil {
		return Call{}, 0, false
	}
	for _, st := range view.Steps {
		if st.NodeID != nodeID {
			continue
		}
		for i, c := range st.Calls {
			if isComponent(in, c.To) {
				return c, i, true
			}
		}
	}
	return Call{}, 0, false
}

// resolveOp returns the contract op whose Normalize(name) == Normalize(label); "" when none.
func resolveOp(in Input, component, label string) string {
	for k, ct := range in.Contracts {
		if Normalize(k) != Normalize(component) {
			continue
		}
		for _, op := range ct.Ops {
			if Normalize(op.Name) == Normalize(label) {
				return op.Name
			}
		}
	}
	return ""
}
```

```go
import (

type graph struct {
	nodes map[string]Node
	out   map[string][]Edge // sorted by (To, Guard)
	in    map[string]int
}

func buildGraph(d Diagram) (graph, error) {
	g := graph{nodes: map[string]Node{}, out: map[string][]Edge{}, in: map[string]int{}}
	for _, n := range d.Nodes {
		if _, dup := g.nodes[n.ID]; dup {
			return g, fmt.Errorf("use case %s: duplicate node id %q", d.UseCaseID, n.ID)
		}
		if !knownKind(n.Kind) {
			return g, fmt.Errorf("use case %s: node %q has unknown kind %q", d.UseCaseID, n.ID, n.Kind)
		}
		g.nodes[n.ID] = n
	}
	for _, e := range d.Edges {
		if _, ok := g.nodes[e.From]; !ok {
			return g, fmt.Errorf("use case %s: edge from unknown node %q", d.UseCaseID, e.From)
		}
		if _, ok := g.nodes[e.To]; !ok {
			return g, fmt.Errorf("use case %s: edge to unknown node %q", d.UseCaseID, e.To)
		}
		g.out[e.From] = append(g.out[e.From], e)
		g.in[e.To]++
	}
	for k := range g.out {
		es := g.out[k]
		sort.SliceStable(es, func(i, j int) bool {
			if es[i].To != es[j].To {
				return es[i].To < es[j].To
			}
			return es[i].Guard < es[j].Guard
		})
	}
	return g, nil
}

func knownKind(k string) bool {
	switch k {
	case "start", "action", "decision", "merge", "fork", "join", "end", "swimLane", "note", "loop",
		"switch", "goto", "interruptEdge", "timeEvent", "acceptEvent", "sendSignal", "objectNode":
		return true
	}
	return false
}

// entries: start nodes, plus timeEvent/acceptEvent nodes with no in-edge. Sorted:
// start first, then by id.
func (g graph) entries() []string {
	var starts, events []string
	for id, n := range g.nodes {
		switch {
		case n.Kind == "start":
			starts = append(starts, id)
		case (n.Kind == "timeEvent" || n.Kind == "acceptEvent") && g.in[id] == 0:
			events = append(events, id)
		}
	}
	sort.Strings(starts)
	sort.Strings(events)
	return append(starts, events...)
}

type rawPath struct {
	nodes  []string
	guards []Guard
}

// elementary enumerates simple paths (no repeated node) from entry to any end
// node, in canonical DFS order. It also records back-edges encountered.
func (g graph) elementary(entry string) (paths []rawPath, backEdges []Edge) {
	onPath := map[string]bool{}
	var walk func(cur string, p rawPath)
	walk = func(cur string, p rawPath) {
		p.nodes = append(append([]string{}, p.nodes...), cur)
		onPath[cur] = true
		defer delete(onPath, cur)
		if g.nodes[cur].Kind == "end" || len(g.out[cur]) == 0 {
			paths = append(paths, p)
			return
		}
		for _, e := range g.out[cur] {
			if onPath[e.To] {
				backEdges = append(backEdges, e)
				continue
			}
			np := rawPath{nodes: p.nodes, guards: append(append([]Guard{}, p.guards...))}
			if e.Guard != "" || isBranch(g.nodes[cur].Kind) {
				np.guards = append(np.guards, Guard{At: cur, Guard: e.Guard})
			}
			walk(e.To, np)
		}
	}
	walk(entry, rawPath{})
	return paths, backEdges
}

func isBranch(kind string) bool {
	return kind == "decision" || kind == "switch" || kind == "interruptEdge" || kind == "loop"
}

// unrollOnce: for each back-edge u->v, take the lowest-ordered elementary path
// containing v before u and emit prefix[..u] + body[v..u] + suffix(after u).
func unrollOnce(paths []rawPath, backEdges []Edge) []rawPath {
	var out []rawPath
	seen := map[string]bool{}
	for _, b := range backEdges {
		key := b.From + "->" + b.To
		if seen[key] {
			continue
		}
		seen[key] = true
		for _, p := range paths {
			vi, ui := index(p.nodes, b.To), index(p.nodes, b.From)
			if vi < 0 || ui < 0 || vi > ui {
				continue
			}
			nodes := append([]string{}, p.nodes[:ui+1]...)
			nodes = append(nodes, p.nodes[vi:ui+1]...)
			nodes = append(nodes, p.nodes[ui+1:]...)
			guards := append([]Guard{}, p.guards...)
			guards = append(guards, Guard{At: b.From, Guard: b.Guard})
			out = append(out, rawPath{nodes: nodes, guards: guards})
			break
		}
	}
	return out
}

func index(s []string, v string) int {
	for i, x := range s {
		if x == v {
			return i
		}
	}
	return -1
}
```

```go
func Derive(in Input) ([]Scenario, error) {

			raws = append(raws, unrollOnce(ps, backs)...)
		}

func title(base string, guards []Guard) string {
	var gs []string
	for _, g := range guards {
		if g.Guard != "" {
			gs = append(gs, g.Guard)
		}
	}
	if len(gs) == 0 {
		return base
	}
	return base + " — " + strings.Join(gs, " · ")
}

// stimuli walks the path, opening a new Stimulus at each input and attaching
// outputs/states/hints to the open one (or to preconditions before the first).
func stimuli(in Input, g graph, d Diagram, view *View, rp rawPath) (Outcome, []Stimulus) {
	var pre Outcome
	var out []Stimulus
	cur := func() *Outcome {
		if len(out) == 0 {
			return &pre
		}
		return &out[len(out)-1].Outcome
	}
	hintsByAnchor := map[string][]string{}
	for _, n := range d.Nodes {
		if n.Kind == "note" && n.Anchor != "" {
			hintsByAnchor[n.Anchor] = append(hintsByAnchor[n.Anchor], n.Label)
		}
	}
	guardAt := map[string]string{}
	for _, gd := range rp.guards {
		guardAt[gd.At] = gd.Guard
	}
	for _, id := range rp.nodes {
		n := g.nodes[id]
		switch classify(n, in, view) {
		case classInput:
			ev := InputEv{Kind: n.Kind}
			if n.Kind == "decision" || n.Kind == "switch" {
				ev.From = n.DecidedBy
				ev.Guard = guardAt[id]
			} else {
				ev.From = n.LinkedActorID
			}
			if c, idx, ok := firstCallInto(in, view, id); ok {
				ev.To = Normalize(c.To)
				ev.Op = resolveOp(in, c.To, c.Label)
				ev.Call = fmt.Sprintf("%s/%s/%d", d.UseCaseID, id, idx)
				if ev.From == "" {
					ev.From = c.From
				}
			} else if lane := componentLane(in, n); lane != "" {
				ev.To = lane
			}
			out = append(out, Stimulus{Seq: len(out) + 1, Node: id, Input: ev})
		case classOutput:
			o := Output{Node: id, Kind: "inferred", To: n.LinkedActorID, Label: n.Label}
			if n.Kind == "sendSignal" {
				o.Kind = "sendSignal"
			}
			c := cur()
			c.ExpectedOutputs = append(c.ExpectedOutputs, o)
		case classState:
			c := cur()
			c.ExpectedStates = append(c.ExpectedStates, State{Node: id, Object: n.Label, InState: n.InState})
		}
		if hs, ok := hintsByAnchor[id]; ok {
			c := cur()
			c.Hints = append(c.Hints, hs...)
		}
	}
	return pre, out
}

// componentLane is the §2.2 rule-5 fallback: a node in a component's lane with no
// dynamic-view call targets that component.
func componentLane(in Input, n Node) string {
	if n.RoleName == "" {
		return ""
	}
	if isComponent(in, n.RoleName) {
		return Normalize(n.RoleName)
	}
	return ""
}

func ForComponent(all []Scenario, component string) []Scenario {
	key := Normalize(component)
	var out []Scenario
	for _, s := range all {
		var kept []Stimulus
		for _, st := range s.Stimuli {
			if st.Input.To == key {
				st.Seq = len(kept) + 1
				kept = append(kept, st)
			}
		}
		if len(kept) == 0 {
			continue
		}
		cp := s
		cp.Stimuli = kept
		out = append(out, cp)
	}
	return out
}
```

```go
func scenarioTestViolations(p *packages.Package, spec Spec, scenariosFile, hooksFile string) []fileLayoutViolation {
	dir := filepath.Dir(p.CompiledGoFiles[0])
	entries, err := os.ReadDir(dir)
	if err != nil {
		return nil
	}
	var out []fileLayoutViolation
	for _, e := range entries {
		name := e.Name()
		if e.IsDir() || !strings.HasSuffix(name, "_test.go") {
			continue
		}
		if name != scenariosFile && name != hooksFile {
			out = append(out, fileLayoutViolation{p.PkgPath, name, "scenario-tests-only", "want only " + scenariosFile + " and " + hooksFile})
			continue
		}
		f, perr := parser.ParseFile(token.NewFileSet(), filepath.Join(dir, name), nil, parser.ImportsOnly)
		if perr != nil {
			out = append(out, fileLayoutViolation{p.PkgPath, name, "scenario-tests-only", "unparseable: " + perr.Error()})
			continue
		}
		if f.Name.Name != p.Name+"_test" {
			out = append(out, fileLayoutViolation{p.PkgPath, name, "test-package-not-external", "want package " + p.Name + "_test"})
```

```go
const (
	ruleTPBound     RuleID = "TP-BOUND"
	ruleTPStep      RuleID = "TP-STEP"
	ruleTPStepOp    RuleID = "TP-STEP-OP"
	ruleTPArgName   RuleID = "TP-ARG-NAME"
	ruleTPArgType   RuleID = "TP-ARG-TYPE"
	ruleTPExpect    RuleID = "TP-EXPECT"
	ruleTPProbe     RuleID = "TP-PROBE"
	ruleTPState     RuleID = "TP-STATE"
	ruleTPOpReached RuleID = "TP-OP-REACHED"
	ruleTPSkip      RuleID = "TP-SKIP"

func testPlanFindings(p Project) ([]Finding, error) {
	all, err := DeriveScenarios(p)
	if err != nil {
		return nil, err
	}
	var out []Finding
	out = append(out, opReachedFindings(p, all)...)
	if p.PhaseArtifacts == nil {
		return out, nil
	}
	for comp, rec := range p.PhaseArtifacts.TestPlan {
		sec := "phaseArtifacts.testPlan." + comp
		l := loc(0, sec)
		projected := scenario.ForComponent(all, comp)
		byID := map[string]scenario.Scenario{}
		for _, s := range projected {
			byID[s.ID] = s
		}
		bound := map[string]bool{}
		for _, b := range rec.Bindings {
			s, ok := byID[b.Scenario]
			if !ok {
				out = append(out, Finding{ruleTPBound, SeverityError, fmt.Sprintf("%s binds unknown scenario %q (stale after a use-case edit?)", comp, b.Scenario), l})
				continue
			}
			if bound[b.Scenario] {
				out = append(out, Finding{ruleTPBound, SeverityError, fmt.Sprintf("%s binds %q twice", comp, b.Scenario), l})
			}
			bound[b.Scenario] = true
			if b.Skip != nil {
				out = append(out, skipFindings(p, comp, b, l)...)
				continue
			}
			out = append(out, stepFindings(p, comp, s, b, l)...)
		}
		for _, s := range projected {
			if !bound[s.ID] {
				out = append(out, Finding{ruleTPBound, SeverityError, fmt.Sprintf("%s has no binding for scenario %s", comp, s.ID), l})
			}
		}
	}
	return out, nil
}

func stepFindings(p Project, comp string, s scenario.Scenario, b ScenarioBinding, l *Location) []Finding {
	var out []Finding
	if len(b.Steps) != len(s.Stimuli) {
		return append(out, Finding{ruleTPStep, SeverityError, fmt.Sprintf("%s %s: %d steps bound, %d stimuli projected", comp, s.ID, len(b.Steps), len(s.Stimuli)), l})
	}
	contract, op := findContract(p, comp)
	for i, st := range b.Steps {
		stim := s.Stimuli[i]
		if st.Seq != stim.Seq {
			out = append(out, Finding{ruleTPStep, SeverityError, fmt.Sprintf("%s %s: step %d has seq %d", comp, s.ID, i+1, st.Seq), l})
		}
		if stim.Input.Op != "" && st.Operation != stim.Input.Op {
			out = append(out, Finding{ruleTPStepOp, SeverityError, fmt.Sprintf("%s %s step %d: operation %q but the call chain resolves %q", comp, s.ID, st.Seq, st.Operation, stim.Input.Op), l})
		}
		cop, ok := op(st.Operation)
		if !ok {
			out = append(out, Finding{ruleTPStepOp, SeverityError, fmt.Sprintf("%s %s step %d: %q is not an operation of %s", comp, s.ID, st.Seq, st.Operation, comp), l})
			continue
		}
		out = append(out, argFindings(st.Inputs, cop, contract, ruleTPArgName, ruleTPArgType, l)...)
		out = append(out, expectFindings(st.Expect, cop, contract, ruleTPExpect, l)...)
		proved := map[string]bool{}
		for _, pr := range st.Probes {
			pc, pop := findContract(p, pr.Component)
			cpop, ok := pop(pr.Operation)
			if !ok {
				out = append(out, Finding{ruleTPProbe, SeverityError, fmt.Sprintf("%s %s step %d: probe %s.%s is not a designed operation", comp, s.ID, st.Seq, pr.Component, pr.Operation), l})
				continue
			}
			out = append(out, argFindings(pr.Inputs, cpop, pc, ruleTPProbe, ruleTPProbe, l)...)
			if pr.For != "" {
				proved[pr.For] = true
			}
		}
		unobs := map[string]bool{}
		for _, u := range st.Unobservable {
			unobs[u] = true
		}
		for _, es := range stim.ExpectedStates {
			if !proved[es.Node] {
				out = append(out, Finding{ruleTPState, SeverityError, fmt.Sprintf("%s %s step %d: expected state %s %s (node %s) has no probe", comp, s.ID, st.Seq, es.Object, es.InState, es.Node), l})
			}
		}
		for _, eo := range stim.ExpectedOutputs {
			if !proved[eo.Node] && !unobs[eo.Node] {
				out = append(out, Finding{ruleTPState, SeverityError, fmt.Sprintf("%s %s step %d: expected output %q (node %s) has neither a probe nor an unobservable declaration", comp, s.ID, st.Seq, eo.Label, eo.Node), l})
			}
		}
	}
	return out
}

func opReachedFindings(p Project, all []scenario.Scenario) []Finding {
	reached := map[string]map[string]bool{}
	for _, s := range all {
		for _, st := range s.Stimuli {
			if st.Input.Op == "" {
				continue
			}
			if reached[st.Input.To] == nil {
				reached[st.Input.To] = map[string]bool{}
			}
			reached[st.Input.To][st.Input.Op] = true
		}
	}
	var out []Finding
	for key, sc := range p.ServiceContracts {
		for _, op := range sc.Interface.Operations {
			if !reached[scenario.Normalize(key)][op.Name] {
				out = append(out, Finding{ruleTPOpReached, SeverityError, fmt.Sprintf("%s.%s is reached by no scenario: dead operation or missing use case", key, op.Name), loc(0, "serviceContracts."+key)})
			}
		}
	}
	return out
}

func skipFindings(p Project, comp string, b ScenarioBinding, l *Location) []Finding {
	if b.Skip.Until == "" || b.Skip.Reason == "" {
		return []Finding{{ruleTPSkip, SeverityError, fmt.Sprintf("%s %s: skip needs reason and until", comp, b.Scenario), l}}
	}
	if integrated(p, b.Skip.Until) {
		return []Finding{{ruleTPSkip, SeverityError, fmt.Sprintf("%s %s: skip waits on %s, which is already integrated", comp, b.Scenario, b.Skip.Until), l}}
	}
	return nil
}
```

```go
func useCaseIORules(c CoreUseCases, actors map[string]bool) []Finding {
	var out []Finding
	for i, uc := range c.UseCases {
		if uc.Activity == nil {
			continue
		}
		l := loc(i, "useCases["+uc.ID+"].activity")
		ids := map[string]bool{}
		for _, n := range uc.Activity.Nodes {
			ids[n.ID] = true
		}
		inputs := 0
		for _, n := range uc.Activity.Nodes {
			switch n.Kind {
			case kindSendSignal:
				if n.LinkedActorID == nil || !actors[*n.LinkedActorID] {
					out = append(out, Finding{ruleUCIOKinds, SeverityError, fmt.Sprintf("sendSignal %q must name a receiving actor", n.ID), l})
				}
			case kindObjectNode:
				if n.InState == "" {
					out = append(out, Finding{ruleUCIOKinds, SeverityError, fmt.Sprintf("objectNode %q must carry inState", n.ID), l})
				}
			case kindNote:
				if n.Anchor != "" && !ids[n.Anchor] {
					out = append(out, Finding{ruleUCIOKinds, SeverityError, fmt.Sprintf("note %q anchors unknown node %q", n.ID, n.Anchor), l})
				}
			case kindAcceptEvent, kindTimeEvent:
				inputs++
			case nodeAction:
				if n.LinkedActorID != nil && actors[*n.LinkedActorID] {
					inputs++
				}
			case nodeDecision, nodeSwitch:
				if n.DecidedBy != nil && actors[*n.DecidedBy] {
					inputs++
				}
			}
		}
		if inputs == 0 {
			out = append(out, Finding{ruleUCNoIO, SeverityWarning, "use case yields no external input stimulus; no scenario can be bound", l})
		}
	}
	return out
}
```

```go
func (c *AppClient) ListArtifacts(ctx context.Context, owner, repo string, runID int64, instToken string) ([]Artifact, error) {
	u := fmt.Sprintf("%s/repos/%s/%s/actions/runs/%d/artifacts?per_page=100", c.baseURL, owner, repo, runID)
	status, body, err := c.do(ctx, http.MethodGet, u, nil, "", instToken)
	if err != nil {
		return nil, err
	}
	if status < 200 || status >= 300 {
		return nil, ClassifyStatus(status, "ListArtifacts")

	}
	out := make([]Artifact, 0, len(d.Artifacts))
	for _, a := range d.Artifacts {

	}
	return out, nil

func (c *AppClient) DownloadArtifact(ctx context.Context, owner, repo string, artifactID int64, instToken string) (io.ReadCloser, error) {
	u := fmt.Sprintf("%s/repos/%s/%s/actions/artifacts/%d/zip", c.baseURL, owner, repo, artifactID)
	req, err := c.newRequest(ctx, http.MethodGet, u, nil, "", instToken)
	if err != nil {
		return nil, err
	}
	client := *c.http
	client.CheckRedirect = func(r *http.Request, via []*http.Request) error {
		if r.URL.Host != via[0].URL.Host {
			r.Header.Del("Authorization")
		}
		if len(via) >= 5 {
			return fwra.New(fwra.Infrastructure, "DownloadArtifact: too many redirects")
		}
		return nil
	}
	resp, err := client.Do(req)
	if err != nil {
		return nil, fwra.Wrap(fwra.Transient, err, "DownloadArtifact")
	}
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		_ = resp.Body.Close()
		return nil, ClassifyStatus(resp.StatusCode, "DownloadArtifact")
	}
	return resp.Body, nil
}
```
