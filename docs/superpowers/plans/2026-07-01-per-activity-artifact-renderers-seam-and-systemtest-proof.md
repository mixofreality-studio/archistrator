# Per-Activity Artifact Renderers — Seam + System Test Proof — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Generalize the webApp artifact panel into a classification-keyed renderer registry and prove it end-to-end with a first-class **System Test** (N-IT) artifact view, on archistrator's own dogfooding state.

**Architecture:** The activity's *classification* (`ActivityType` + `TestingVariant`) is derived in the Go view-model mapper from the activity id (the existing `DeriveType`/`DeriveVariant`), carried to the SPA, and used as the dispatch key for a `artifactRenderers` registry. Each type is a thin renderer composed from small primitives; a fallback preserves today's honest-pointer card. The proof type (System Test) reads the already-modeled `TestingState` (runs + defects).

**Tech Stack:** Go (server, `cmd/modelgen`/`cmd/clientgen` codegen from `.aiarch/state/project.json`), React + MUI + `@xyflow/react` (webApp), Playwright (`uitests/`, black-box against the running SPA).

```go
// testingStateToContract converts the head-state TestingState to the contract
// view. Returns nil when absent so the field is omitted from the read.
func testingStateToContract(ts *projectstate.TestingState) *TestingStateView {
	if ts == nil {
		return nil
	}
	runs := make([]TestRunView, len(ts.TestRuns))
	for i, r := range ts.TestRuns {
		runs[i] = TestRunView{ID: r.ID, Passed: r.Passed, Failed: r.Failed, Note: r.Note}
	}
	defects := make([]DefectView, len(ts.Defects))
	for i, d := range ts.Defects {
		defects[i] = DefectView{ID: d.ID, Title: d.Title, Severity: d.Severity, Note: d.Note}
	}
	return &TestingStateView{TestRuns: runs, Defects: defects}
}
```

```tsx
import type { ReactNode } from 'react';
import Box from '@mui/material/Box';
import Typography from '@mui/material/Typography';
import Chip from '@mui/material/Chip';
import { UI_IDENTIFIERS } from '../../../constants/UIIdentifiers';
import type { ArtifactRendererProps } from '../artifactRenderers';
import { StatTile } from '../primitives/StatTile';
import { RecordTable } from '../primitives/RecordTable';

export function SystemTestView({ project, t }: ArtifactRendererProps): ReactNode {
  const ts = project?.testingState;
  const runs = ts?.testRuns ?? [];
  const defects = ts?.defects ?? [];
  const totalPassed = runs.reduce((n, r) => n + r.passed, 0);
  const totalFailed = runs.reduce((n, r) => n + r.failed, 0);
  const openDefects = defects.length;

  if (ts === undefined || (runs.length === 0 && defects.length === 0)) {
    return (
      <Typography sx={{ color: t.muted, fontSize: 12.5 }} data-testid={UI_IDENTIFIERS.Construction.SYSTEM_TEST_VIEW}>
        No system-test runs recorded yet. Runs and defects appear here once the System Test activity (N-IT) executes.
      </Typography>
    );
  }

  const sevTone = (s: string): 'bad' | 'neutral' =>
    s === 'critical' || s === 'high' ? 'bad' : 'neutral';

  return (
    <Box
      data-testid={UI_IDENTIFIERS.Construction.SYSTEM_TEST_VIEW}
      sx={{ display: 'flex', flexDirection: 'column', gap: 1.5 }}
    >
      <Typography sx={{ fontFamily: t.mono, fontWeight: 700, fontSize: 11, letterSpacing: '0.06em', color: t.ink }}>
        SYSTEM TEST · {runs.length} run{runs.length === 1 ? '' : 's'}
      </Typography>
      <Box sx={{ display: 'flex', gap: 1, flexWrap: 'wrap' }}>
        <StatTile label="passed" value={totalPassed} tone="good" t={t} />
        <StatTile label="failed" value={totalFailed} tone={totalFailed > 0 ? 'bad' : 'good'} t={t} />
        <StatTile label="open defects" value={openDefects} tone={openDefects > 0 ? 'bad' : 'good'} t={t} />
      </Box>

      {runs.length > 0 ? (
        <RecordTable
          columns={[
            { key: 'id', label: 'Run' },
            { key: 'passed', label: 'Passed' },
            { key: 'failed', label: 'Failed' },
            { key: 'note', label: 'Note' },
          ]}
          rows={runs.map((r) => ({ id: r.id, passed: r.passed, failed: r.failed, note: r.note }))}
          t={t}
        />
      ) : null}

      {defects.length > 0 ? (
        <>
          <Typography sx={{ fontFamily: t.mono, fontWeight: 700, fontSize: 11, letterSpacing: '0.06em', color: t.ink }}>
            DEFECTS
          </Typography>
          <RecordTable
            columns={[
              { key: 'id', label: 'ID' },
              { key: 'title', label: 'Title' },
              { key: 'severity', label: 'Severity' },
              { key: 'note', label: 'Note' },
            ]}
            rows={defects.map((d) => ({
              id: d.id,
              title: d.title,
              severity: (
                <Chip
                  label={d.severity}
                  size="small"
                  sx={{
                    height: 18,
                    fontSize: 8.5,
                    bgcolor: sevTone(d.severity) === 'bad' ? t.awaitingBg : t.paperAlt,
                    color: sevTone(d.severity) === 'bad' ? t.awaitingFg : t.muted,
                  }}
                />
              ),
              note: d.note,
            }))}
            t={t}
          />
        </>
      ) : null}
    </Box>
  );
}
```
